const canvas = document.getElementById("bgCanvas");
const ctx = canvas.getContext("2d");

let particles = [];
let loveParticles = [];
let modeLove = false;

function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
}
resizeCanvas();
window.addEventListener("resize", resizeCanvas);

class Particle {
    constructor(x, y, speedX, speedY, size) {
        this.x = x;
        this.y = y;
        this.speedX = speedX;
        this.speedY = speedY;
        this.size = size;
    }
    update() {
        this.x += this.speedX;
        this.y += this.speedY;

        if (this.x > canvas.width || this.x < 0) this.speedX *= -1;
        if (this.y > canvas.height || this.y < 0) this.speedY *= -1;
    }
    draw() {
        ctx.fillStyle = "rgba(0,180,255,0.5)";
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
        ctx.fill();
    }
}

function initParticles() {
    particles = [];
    for (let i = 0; i < 80; i++) {
        particles.push(
            new Particle(
                Math.random() * canvas.width,
                Math.random() * canvas.height,
                (Math.random() - 0.5) * 0.5,
                (Math.random() - 0.5) * 0.5,
                Math.random() * 2 + 1
            )
        );
    }
}
initParticles();

function connectParticles() {
    for (let a = 0; a < particles.length; a++) {
        for (let b = a; b < particles.length; b++) {
            let dx = particles[a].x - particles[b].x;
            let dy = particles[a].y - particles[b].y;
            let distance = dx * dx + dy * dy;

            if (distance < 10000) {
                ctx.strokeStyle = "rgba(0,150,255,0.08)";
                ctx.lineWidth = 1;
                ctx.beginPath();
                ctx.moveTo(particles[a].x, particles[a].y);
                ctx.lineTo(particles[b].x, particles[b].y);
                ctx.stroke();
            }
        }
    }
}

/* LOVE MODE PARTICLES */
function createLoveBurst() {
    loveParticles = [];
    for (let i = 0; i < 50; i++) {
        loveParticles.push({
            x: canvas.width / 2,
            y: canvas.height / 2,
            size: Math.random() * 6 + 4,
            speedX: (Math.random() - 0.5) * 4,
            speedY: (Math.random() - 0.5) * 4,
            alpha: 1
        });
    }
}

function drawLoveParticles() {
    loveParticles.forEach(p => {
        p.x += p.speedX;
        p.y += p.speedY;
        p.alpha -= 0.01;

        ctx.fillStyle = `rgba(255,80,120,${p.alpha})`;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
        ctx.fill();
    });

    loveParticles = loveParticles.filter(p => p.alpha > 0);
}

function animate() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    particles.forEach(p => {
        p.update();
        p.draw();
    });

    connectParticles();

    if (modeLove) {
        drawLoveParticles();
    }

    requestAnimationFrame(animate);
}
animate();

/* TRIGGER LOVE MODE (dipanggil saat klimaks pesan) */
function activateLoveMode() {
    modeLove = true;
    document.body.classList.add("love-mode");
    createLoveBurst();
}

/* RESET MODE (ketika Content diklik) */
function resetLoveMode() {
    modeLove = false;
    document.body.classList.remove("love-mode");
    loveParticles = [];
}
